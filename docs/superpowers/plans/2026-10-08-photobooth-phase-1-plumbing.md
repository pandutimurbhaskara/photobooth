`src/shared/room-code.ts`:
```ts
import { ROOM_CODE_ALPHABET, ROOM_CODE_LENGTH } from './constants'

const SIZE = ROOM_CODE_ALPHABET.length
// Largest multiple of SIZE that fits in a byte; bytes at or above it are discarded
// so `byte % SIZE` is uniform (rejection sampling).
const LIMIT = 256 - (256 % SIZE)
const VALID = new RegExp(`^[${ROOM_CODE_ALPHABET}]{${ROOM_CODE_LENGTH}}$`)

export function generateRoomCode(
  random: (buf: Uint8Array) => Uint8Array = (buf) => crypto.getRandomValues(buf),
): string {
  let out = ''
  while (out.length < ROOM_CODE_LENGTH) {
    for (const byte of random(new Uint8Array(16))) {
      if (byte < LIMIT && out.length < ROOM_CODE_LENGTH) out += ROOM_CODE_ALPHABET[byte % SIZE]
    }
  }
  return out
}

export function isValidRoomCode(value: unknown): value is string {
  return typeof value === 'string' && VALID.test(value)
}

export function normalizeRoomInput(input: string): string | null {
  const fromUrl = /\/b\/([^/?#\s]+)/i.exec(input)
  const code = (fromUrl?.[1] ?? input).toUpperCase().replace(/[\s-]/g, '')
  return isValidRoomCode(code) ? code : null
}

export function formatRoomCode(code: string): string {
  return `${code.slice(0, 3)} ${code.slice(3)}`
}
```

```ts
  | { type: 'answer'; answer: AnswerInit }

  | { type: 'capture-scheduled'; captureAt: number; round: number }
  | { type: 'pong'; clientTime: number; serverTime: number }
  | { type: 'error'; code: ErrorCode }

type Rec = Record<string, unknown>
const isRec = (x: unknown): x is Rec => typeof x === 'object' && x !== null && !Array.isArray(x)
const onlyKeys = (o: Rec, keys: readonly string[]): boolean => Object.keys(o).every((k) => keys.includes(k))
const isRound = (v: unknown): v is number => typeof v === 'number' && Number.isInteger(v) && v >= 0 && v <= 1_000_000
const isOptStr = (v: unknown): v is string | null | undefined =>
  v === undefined || v === null || (typeof v === 'string' && v.length <= 256)
const isOptIdx = (v: unknown): v is number | null | undefined =>
  v === undefined || v === null || (typeof v === 'number' && Number.isInteger(v) && v >= 0 && v < 64)

function sdp<T extends 'offer' | 'answer'>(v: unknown, type: T): { type: T; sdp: string } | null {
  if (!isRec(v) || !onlyKeys(v, ['type', 'sdp'])) return null
  const body = v.sdp
  if (v.type !== type || typeof body !== 'string' || body.length === 0 || body.length > MAX_SDP_CHARS) return null
  return { type, sdp: body }
}

function candidate(v: unknown): IceCandidateInit | null {
  if (!isRec(v) || !onlyKeys(v, ['candidate', 'sdpMid', 'sdpMLineIndex', 'usernameFragment'])) return null
  const { candidate: text, sdpMid, sdpMLineIndex, usernameFragment } = v
  if (typeof text !== 'string' || text.length > MAX_CANDIDATE_CHARS) return null
  if (!isOptStr(sdpMid) || !isOptIdx(sdpMLineIndex) || !isOptStr(usernameFragment)) return null
  const out: IceCandidateInit = { candidate: text }
  if (sdpMid !== undefined) out.sdpMid = sdpMid
  if (sdpMLineIndex !== undefined) out.sdpMLineIndex = sdpMLineIndex
  if (usernameFragment !== undefined) out.usernameFragment = usernameFragment
  return out
}

export function parseClientMessage(raw: unknown): ClientMessage | null {
  if (!isRec(raw) || typeof raw.type !== 'string') return null
  switch (raw.type) {
    case 'join':
      return onlyKeys(raw, ['type', 'participantId']) &&
        typeof raw.participantId === 'string' &&
        PARTICIPANT_ID_RE.test(raw.participantId)
        ? { type: 'join', participantId: raw.participantId }
        : null
    case 'ready':
      return onlyKeys(raw, ['type']) ? { type: 'ready' } : null
    case 'unready':
      return onlyKeys(raw, ['type']) ? { type: 'unready' } : null
    case 'offer': {
      const offer = onlyKeys(raw, ['type', 'offer']) ? sdp(raw.offer, 'offer') : null
      return offer ? { type: 'offer', offer } : null
    }
    case 'answer': {
      const answer = onlyKeys(raw, ['type', 'answer']) ? sdp(raw.answer, 'answer') : null
      return answer ? { type: 'answer', answer } : null
    }
    case 'ice-candidate': {
      const c = onlyKeys(raw, ['type', 'candidate']) ? candidate(raw.candidate) : null
      return c ? { type: 'ice-candidate', candidate: c } : null
    }
    case 'photo-complete':
      return onlyKeys(raw, ['type', 'round']) && isRound(raw.round)
        ? { type: 'photo-complete', round: raw.round }
        : null
    case 'ping':
      return onlyKeys(raw, ['type', 'clientTime']) &&
        typeof raw.clientTime === 'number' &&
        Number.isFinite(raw.clientTime)
        ? { type: 'ping', clientTime: raw.clientTime }
        : null
    default:
      return null
  }
}

export function decodeClientFrame(data: unknown): ClientMessage | null {
  if (typeof data !== 'string' || data.length > MAX_WS_FRAME_BYTES) return null
  try {
    return parseClientMessage(JSON.parse(data))
  } catch {
    return null
  }
}
```

```ts
import { env } from 'cloudflare:workers'
import { generateRoomCode } from '../shared/room-code'
import type { ClientMessage, ServerMessage } from '../shared/protocol'

const sleep = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms))
const open: TestClient[] = []

export class TestClient {
  messages: ServerMessage[] = []
  closeCode: number | null = null
  private wake: (() => void) | null = null

  constructor(readonly ws: WebSocket) {
    ws.addEventListener('message', (e) => {
      this.messages.push(JSON.parse(e.data as string) as ServerMessage)
      this.wake?.()
    })
    ws.addEventListener('close', (e) => {
      this.closeCode = e.code
      this.wake?.()
    })
    open.push(this)
  }

  send(message: ClientMessage | Record<string, unknown>): void {
    this.ws.send(JSON.stringify(message))
  }
  sendRaw(text: string): void {
    this.ws.send(text)
  }
  clear(): void {
    this.messages.length = 0
  }

  private async until<T>(probe: () => T | undefined, what: string, ms: number): Promise<T> {
    const end = Date.now() + ms
    for (;;) {
      const value = probe()
      if (value !== undefined) return value
      if (Date.now() > end) {
        throw new Error(`timeout waiting for ${what}; queue=${JSON.stringify(this.messages)} close=${this.closeCode}`)
      }
      await Promise.race([new Promise<void>((resolve) => (this.wake = resolve)), sleep(25)])
    }
  }

  /** Remove and return the first queued message of `type`, waiting for it if needed. */
  take<T extends ServerMessage['type']>(type: T, ms = 2000): Promise<Extract<ServerMessage, { type: T }>> {
    return this.until(() => {
      const i = this.messages.findIndex((m) => m.type === type)
      return i < 0 ? undefined : (this.messages.splice(i, 1)[0] as Extract<ServerMessage, { type: T }>)
    }, type, ms)
  }
  waitClose(ms = 2000): Promise<number> {
    return this.until(() => this.closeCode ?? undefined, 'close', ms)
  }
  async expectNone(type: ServerMessage['type'], ms = 150): Promise<void> {
    await sleep(ms)
    if (this.messages.some((m) => m.type === type)) throw new Error(`unexpected "${type}" in ${JSON.stringify(this.messages)}`)
  }
}
```

```ts
  ROOM_IDLE_MS,
  SLOT_GRACE_MS,

import type { ErrorCode, Phase, PState, RoomState, ServerMessage, SlotName } from '../shared/protocol'

const WS_OPEN = 1
const SLOTS: readonly SlotName[] = ['a', 'b']

interface Slot {
  participantId: string
  ready: boolean
  photoDone: boolean
  lastSeenAt: number
}
interface RoomRecord {
  code: string
  createdAt: number
  lastActivityAt: number
  round: number
  captureAt: number | null
  slots: Partial<Record<SlotName, Slot>>
}
interface Attachment {
  slot: SlotName

/**
 * One booth. Hibernating WebSocket server: carries signaling, ready state, clock sync and
 * capture scheduling only — never media, never photos. The persisted record holds no names,
 * IPs or media; `participantId` is the membership secret and is never sent to the partner.
 */
export class BoothRoom extends DurableObject<Env> {
  private rec: RoomRecord | null = null
  private persistedActivityAt = 0
  /** Sockets we know are going away (replaced or closed): excluded from "connected". */
  private readonly closing = new WeakSet<WebSocket>()

  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env)
    void ctx.blockConcurrencyWhile(async () => {
      this.rec = (await ctx.storage.get<RoomRecord>('room')) ?? null
    })
  }

  // ---- RPC (called from the Worker) -------------------------------------------------------

  async create(code: string): Promise<boolean> {
    if (this.rec) return false
    const now = Date.now()
    this.rec = { code, createdAt: now, lastActivityAt: now, round: 0, captureAt: null, slots: {} }
    this.save()
    await this.ctx.storage.setAlarm(now + ROOM_IDLE_MS)
    return true
  }

  async status(participantId: string | null): Promise<{ exists: boolean; full: boolean }> {
    if (!this.rec) return { exists: false, full: false }
    const isMember = participantId !== null && this.slotOf(participantId) !== null
    return { exists: true, full: !isMember && this.freeSlot(Date.now()) === null }
  }

  // ---- WebSocket upgrade ------------------------------------------------------------------

  async fetch(request: Request): Promise<Response> {
    if (!this.rec) return new Response('not found', { status: 404 })
    if (request.headers.get('Upgrade') !== 'websocket') return new Response('expected websocket', { status: 426 })
    if (this.ctx.getWebSockets().length >= MAX_SOCKETS) return new Response('busy', { status: 429 })
    const pair = new WebSocketPair()
    const [client, server] = Object.values(pair)
    this.ctx.acceptWebSocket(server)
    return new Response(null, { status: 101, webSocket: client })
  }

  // ---- Hibernation handlers ---------------------------------------------------------------

  async webSocketMessage(ws: WebSocket, data: string | ArrayBuffer): Promise<void> {
    if (!this.rec) return this.reject(ws, 'room-not-found', CLOSE.NOT_FOUND)
    const msg = decodeClientFrame(data)
    if (msg === null) return this.reject(ws, 'invalid-message')
    const att = ws.deserializeAttachment() as Attachment | null
    if (msg.type === 'join') {
      if (att) return this.reject(ws, 'invalid-message')
      return this.handleJoin(ws, msg.participantId)
    }
    if (!att) return this.reject(ws, 'not-joined')
    this.touch(Date.now())
  }

  async webSocketClose(ws: WebSocket, code: number): Promise<void> {
    this.onGone(ws)
    try {
      ws.close(code === 1000 || (code >= 3000 && code <= 4999) ? code : 1000, 'bye')
    } catch {
      // already closed
    }
  }

  async webSocketError(ws: WebSocket): Promise<void> {
    this.onGone(ws)
  }

  // ---- Join / leave -----------------------------------------------------------------------

  private handleJoin(ws: WebSocket, participantId: string): void {
    const rec = this.room
    const now = Date.now()
    let slot = this.slotOf(participantId)
    if (slot === null) {
      slot = this.freeSlot(now)
      if (slot === null) return this.reject(ws, 'room-full', CLOSE.FULL)
      rec.slots[slot] = { participantId, ready: false, photoDone: false, lastSeenAt: now }
    }
    ws.serializeAttachment({ slot } satisfies Attachment)
    for (const live of this.live()) {
      if (live.slot === slot && live.ws !== ws) {
        this.closing.add(live.ws) // replaced: its close event must not read as a departure
        live.ws.close(CLOSE.REPLACED, 'replaced')
      }
    }
    const me = rec.slots[slot]
    if (me) me.lastSeenAt = now
    this.touch(now)
    this.save()

    this.send(ws, { type: 'room-state', state: this.stateFor(slot, now) }) // join snapshot is always first
    this.sendTo(otherSlot(slot), { type: 'partner-joined' })
    for (const other of this.live()) {
      if (other.ws !== ws) this.send(other.ws, { type: 'room-state', state: this.stateFor(other.slot, now) })
    }
  }

  private onGone(ws: WebSocket): void {
    if (this.closing.has(ws)) return
    this.closing.add(ws)
    const rec = this.rec
    const att = ws.deserializeAttachment() as Attachment | null
    if (!rec || !att) return
    if (this.live().some((s) => s.slot === att.slot)) return // another live socket holds this slot
    const me = rec.slots[att.slot]
    if (!me) return
    me.ready = false
    me.lastSeenAt = Date.now()
    rec.captureAt = null // a round in flight is cancelled when anyone leaves
    this.save()
    this.sendTo(otherSlot(att.slot), { type: 'partner-left' })
    this.broadcastState()
  }

  // ---- State helpers ----------------------------------------------------------------------

  private get room(): RoomRecord {
    if (!this.rec) throw new Error('room was never created')
    return this.rec
  }

  private save(): void {
    void this.ctx.storage.put('room', this.rec)
  }

  private touch(now: number): void {
    this.room.lastActivityAt = now
    if (now - this.persistedActivityAt >= ACTIVITY_PERSIST_MS) {
      this.persistedActivityAt = now
      this.save()
    }
  }

  private slotOf(participantId: string): SlotName | null {
    for (const name of SLOTS) if (this.rec?.slots[name]?.participantId === participantId) return name
    return null
  }

  /** A slot a *new* participant may take: empty, or its owner has been gone past the grace period. */
  private freeSlot(now: number): SlotName | null {
    const connected = this.connected()
    for (const name of SLOTS) {
      const s = this.rec?.slots[name]
      if (!s) return name
      if (!connected.has(name) && now - s.lastSeenAt > SLOT_GRACE_MS) return name
    }
    return null
  }

  private live(): Array<{ ws: WebSocket; slot: SlotName }> {
    const out: Array<{ ws: WebSocket; slot: SlotName }> = []
    for (const ws of this.ctx.getWebSockets()) {
      if (ws.readyState !== WS_OPEN || this.closing.has(ws)) continue
      const att = ws.deserializeAttachment() as Attachment | null
      if (att) out.push({ ws, slot: att.slot })
    }
    return out
  }

  private connected(): Set<SlotName> {
    return new Set(this.live().map((s) => s.slot))
  }

  private phase(now: number): Phase {
    const rec = this.room
    if (rec.captureAt !== null && now >= rec.captureAt + EXCHANGE_TIMEOUT_MS) {
      rec.captureAt = null
      this.save()
    }
    if (this.connected().size < 2) return 'waiting'
    return rec.captureAt !== null ? 'capture' : 'ready'
  }

  private stateFor(slot: SlotName, now: number): RoomState {
    const rec = this.room
    const phase = this.phase(now) // may clear an expired capture, so compute before reading captureAt
    const connected = this.connected()
    const p = (name: SlotName): PState | null => {
      const s = rec.slots[name]
      return s ? { connected: connected.has(name), ready: s.ready, photoDone: s.photoDone } : null
    }
    return { code: rec.code, you: slot, phase, round: rec.round, captureAt: rec.captureAt, participants: { a: p('a'), b: p('b') } }
  }

  // ---- Sending ----------------------------------------------------------------------------

  private send(ws: WebSocket, msg: ServerMessage): void {
    try {
      ws.send(JSON.stringify(msg))
    } catch {
      // socket is closing; its close handler will clean up
    }
  }

  private sendTo(slot: SlotName, msg: ServerMessage): void {
    for (const s of this.live()) if (s.slot === slot) this.send(s.ws, msg)
  }

  private broadcastState(): void {
    const now = Date.now()
    for (const s of this.live()) this.send(s.ws, { type: 'room-state', state: this.stateFor(s.slot, now) })
  }

  private reject(ws: WebSocket, code: ErrorCode, closeCode: number = CLOSE.INVALID): void {
    this.send(ws, { type: 'error', code })
    try {
      ws.close(closeCode, code)
    } catch {
      // already closed
    }
  }
}
```

```ts
/** Membership secret for one tab in one booth: survives reloads, never leaves this browser except to the server. */
export function getParticipantId(code: string): string {
  const key = `booth:pid:${code}`
  try {
    const stored = sessionStorage.getItem(key)
    if (stored) return stored
    const fresh = crypto.randomUUID()
    sessionStorage.setItem(key, fresh)

    // Storage blocked (private mode, embedded browser): keep it for this page's lifetime only.
    const existing = memory.get(key)
    if (existing) return existing
    const fresh = crypto.randomUUID()
    memory.set(key, fresh)
    return fresh
  }
```

```ts
import { CLOSE } from '../../shared/constants'
import type { ClientMessage, ServerMessage } from '../../shared/protocol'

export type SocketStatus = 'connecting' | 'open' | 'reconnecting' | 'closed'

export interface RoomSocketOptions {
  url: string
  participantId: string
  /** `isSnapshot` is true for the first room-state after every (re)open (the join snapshot). */
  onMessage: (message: ServerMessage, isSnapshot: boolean) => void
  onStatus: (status: SocketStatus, closeCode: number | null) => void
  WebSocketImpl?: typeof WebSocket
  backoffMs?: readonly number[]
  giveUpAfterMs?: number
  now?: () => number
}

const TERMINAL = new Set<number>(Object.values(CLOSE))
const WS_OPEN = 1

/** Reconnecting WebSocket for one booth. Rejoins with the same participantId; the server swaps the old socket out. */
export class RoomSocket {
  private ws: WebSocket | null = null
  private attempt = 0
  private downSince: number | null = null
  private timer: ReturnType<typeof setTimeout> | null = null
  private stopped = false
  private snapshotPending = true

  constructor(private readonly o: RoomSocketOptions) {
    this.o.onStatus('connecting', null)
    this.connect()
  }

  send(message: ClientMessage): boolean {
    if (this.ws?.readyState !== WS_OPEN) return false
    this.ws.send(JSON.stringify(message))
    return true
  }

  /** Close and rejoin right now (also restarts a socket that gave up). This is what RECONNECT buttons call. */
  reconnect(): void {
    this.clearTimer()
    this.detach()
    this.stopped = false
    this.attempt = 0
    this.downSince = null
    this.o.onStatus('reconnecting', null)
    this.connect()
  }

  close(): void {
    this.stopped = true
    this.clearTimer()
    this.detach()
  }

  private connect(): void {
    const Impl = this.o.WebSocketImpl ?? WebSocket
    this.snapshotPending = true
    const ws = new Impl(this.o.url)
    this.ws = ws
    ws.addEventListener('open', () => {
      if (this.ws !== ws) return
      this.attempt = 0
      this.downSince = null
      this.o.onStatus('open', null)
      ws.send(JSON.stringify({ type: 'join', participantId: this.o.participantId } satisfies ClientMessage))
    })
    ws.addEventListener('message', (e) => {
      if (this.ws !== ws || typeof e.data !== 'string') return
      let message: ServerMessage
      try {
        message = JSON.parse(e.data) as ServerMessage
      } catch {
        return
      }
      const isSnapshot = this.snapshotPending && message.type === 'room-state'
      if (isSnapshot) this.snapshotPending = false
      this.o.onMessage(message, isSnapshot)
    })
    ws.addEventListener('close', (e) => {
      if (this.ws !== ws) return
      this.ws = null
      this.onClosed(e.code)
    })
  }

  private onClosed(code: number): void {
    if (this.stopped) return
    if (TERMINAL.has(code)) {
      this.stopped = true
      this.o.onStatus('closed', code)
      return
    }
    const now = (this.o.now ?? Date.now)()
    this.downSince ??= now
    if (now - this.downSince > (this.o.giveUpAfterMs ?? 60_000)) {
      this.stopped = true
      this.o.onStatus('closed', code)
      return
    }
    const delays = this.o.backoffMs ?? [500, 1000, 2000, 4000, 5000]
    const delay = delays[Math.min(this.attempt, delays.length - 1)] ?? 5000
    this.attempt += 1
    this.o.onStatus('reconnecting', null)
    this.timer = setTimeout(() => {
      this.timer = null
      this.connect()
    }, delay)
  }

  private detach(): void {
    const ws = this.ws
    this.ws = null
    if (!ws) return
    try {
      ws.close(1000)
    } catch {
      // already closed
    }
  }

  private clearTimer(): void {
    if (this.timer !== null) clearTimeout(this.timer)
    this.timer = null
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react'
import type { ClientMessage, RoomState, ServerMessage } from '../../shared/protocol'
import { RoomSocket, type SocketStatus } from '../lib/room-socket'
import { useLatest } from './useLatest'

interface Options {
  code: string
  participantId: string
  enabled: boolean
  onMessage: (message: ServerMessage, isSnapshot: boolean) => void
}

export function useRoom({ code, participantId, enabled, onMessage }: Options) {
  const [status, setStatus] = useState<SocketStatus>('connecting')
  const [closeCode, setCloseCode] = useState<number | null>(null)
  const [state, setState] = useState<RoomState | null>(null)
  const socketRef = useRef<RoomSocket | null>(null)
  const latest = useLatest(onMessage)

  useEffect(() => {
    if (!enabled) return
    const scheme = location.protocol === 'https:' ? 'wss' : 'ws'
    const socket = new RoomSocket({
      url: `${scheme}://${location.host}/api/rooms/${code}/ws`,
      participantId,
      onStatus: (s, c) => {
        setStatus(s)
        setCloseCode(c)
      },
      onMessage: (m, isSnapshot) => {
        if (m.type === 'room-state') setState(m.state)
        latest.current(m, isSnapshot)
      },
    })
    socketRef.current = socket
    return () => {
      socket.close()
      socketRef.current = null
    }
  }, [enabled, code, participantId, latest])

  const send = useCallback((message: ClientMessage): boolean => socketRef.current?.send(message) ?? false, [])
  const reconnect = useCallback(() => socketRef.current?.reconnect(), [])
  return { status, closeCode, state, send, reconnect }
}
```

```ts
export type Gate = 'checking' | 'ok' | 'not-found' | 'full' | 'error'

/** Pre-flight: does this booth exist, and is there room for *me* (a member reloading is never "full")? */
export function useRoomGate(code: string, participantId: string): Gate {
  const [gate, setGate] = useState<Gate>('checking')
  useEffect(() => {
    const abort = new AbortController()
    fetch(`/api/rooms/${code}`, { headers: { 'x-participant-id': participantId }, signal: abort.signal })
      .then(async (res) => {
        if (res.status === 404) return setGate('not-found')
        if (!res.ok) return setGate('error')
        const body = (await res.json()) as { full: boolean }
        setGate(body.full ? 'full' : 'ok')
      })
      .catch((e: unknown) => {
        if (!(e instanceof DOMException && e.name === 'AbortError')) setGate('error')
      })
    return () => abort.abort()
  }, [code, participantId])
  return gate
}
```

```ts
export type CameraError = 'blocked' | 'unavailable' | 'unsupported'

export function classifyCameraError(e: unknown): CameraError {
  const name = typeof e === 'object' && e !== null && 'name' in e ? String((e as { name: unknown }).name) : ''
  if (name === 'NotAllowedError' || name === 'SecurityError' || name === 'PermissionDeniedError') return 'blocked'
  if (name === 'UnsupportedError') return 'unsupported'
  return 'unavailable'
}

/** The preview is mirrored exactly when the captured photo is (spec D15). Rear cameras are never mirrored. */
export function isMirrored(track: Pick<MediaStreamTrack, 'getSettings'>): boolean {
  return track.getSettings().facingMode !== 'environment'
}

export async function openCamera(deviceId?: string): Promise<MediaStream> {
  if (!navigator.mediaDevices?.getUserMedia) {
    // Insecure context (plain http on a LAN address) hides mediaDevices entirely.
    throw Object.assign(new Error('camera API unavailable'), { name: 'UnsupportedError' })
  }
  return navigator.mediaDevices.getUserMedia({
    audio: false,
    video: {
      width: { ideal: 1280 },
      height: { ideal: 720 },
      ...(deviceId ? { deviceId: { exact: deviceId } } : { facingMode: 'user' }),
    },
  })
}

export async function listVideoInputs(): Promise<MediaDeviceInfo[]> {
  const all = await navigator.mediaDevices.enumerateDevices()
  return all.filter((d) => d.kind === 'videoinput')
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react'
import { classifyCameraError, isMirrored, listVideoInputs, openCamera, type CameraError } from '../lib/camera'

export function useCamera() {
  const [stream, setStream] = useState<MediaStream | null>(null)
  const [error, setError] = useState<CameraError | null>(null)
  const [canFlip, setCanFlip] = useState(false)
  const streamRef = useRef<MediaStream | null>(null)

  const stopCurrent = useCallback(() => {
    streamRef.current?.getTracks().forEach((t) => t.stop())
  }, [])

  const adopt = useCallback((next: MediaStream) => {
    streamRef.current = next
    setStream(next)
  }, [])

  const start = useCallback(async () => {
    setError(null)
    try {
      stopCurrent()
      adopt(await openCamera())
      setCanFlip((await listVideoInputs()).length > 1) // device labels/ids are only exposed after permission
    } catch (e) {
      setError(classifyCameraError(e))
    }
  }, [adopt, stopCurrent])

  /** Cycle to the next camera. The old tracks are stopped first: many phones allow only one open camera. */
  const flip = useCallback(async () => {
    const inputs = await listVideoInputs()
    if (inputs.length < 2) return
    const current = streamRef.current?.getVideoTracks()[0]?.getSettings().deviceId
    const next = inputs[(inputs.findIndex((d) => d.deviceId === current) + 1) % inputs.length]
    if (!next) return
    stopCurrent()
    try {
      adopt(await openCamera(next.deviceId))
    } catch {
      await start()
    }
  }, [adopt, start, stopCurrent])

  useEffect(() => stopCurrent, [stopCurrent])

  const track = stream?.getVideoTracks()[0]
  return { stream, error, mirrored: track ? isMirrored(track) : true, canFlip, start, flip }
}
```

```ts
import type { ClientMessage, IceCandidateInit, IceServerConfig, ServerMessage } from '../../shared/protocol'

export type PeerState = 'connecting' | 'connected' | 'reconnecting' | 'failed' | 'closed'
export type SignalOut = Extract<ClientMessage, { type: 'offer' | 'answer' | 'ice-candidate' }>
export type SignalIn = Extract<ServerMessage, { type: 'offer' | 'answer' | 'ice-candidate' }>

export interface PeerOptions {
  iceServers: IceServerConfig[]
  localStream: MediaStream | null
  signal: (message: SignalOut) => void
  onRemoteStream: (stream: MediaStream) => void
  onState: (state: PeerState) => void
  onChannelOpen: () => void
  onChannelText: (text: string) => void
  onChannelBinary: (data: ArrayBuffer) => void
  PeerConnectionImpl?: typeof RTCPeerConnection
  /** How long a `disconnected` connection may take to recover before we call it failed. */
  failAfterMs?: number
  /** How long the first connection may take before we call it failed. */
  connectTimeoutMs?: number
}

const LOW_WATER = 256 * 1024

/**
 * One RTCPeerConnection plus the photo DataChannel. Slot `a` calls `start()` (offer); slot `b` just
 * feeds `handleSignal()`. There is no renegotiation: a failure is handled by building a new Peer.
 */
export class Peer {
  private readonly pc: RTCPeerConnection
  private readonly channel: RTCDataChannel
  private pending: IceCandidateInit[] = []
  private failTimer: ReturnType<typeof setTimeout> | null = null
  private closed = false

  constructor(private readonly o: PeerOptions) {
    const Impl = o.PeerConnectionImpl ?? RTCPeerConnection
    this.pc = new Impl({ iceServers: o.iceServers })

    this.channel = this.pc.createDataChannel('photos', { negotiated: true, id: 0, ordered: true })
    this.channel.binaryType = 'arraybuffer'
    this.channel.bufferedAmountLowThreshold = LOW_WATER
    this.channel.onopen = () => o.onChannelOpen()
    this.channel.onmessage = (e: MessageEvent) => {
      if (typeof e.data === 'string') o.onChannelText(e.data)
      else o.onChannelBinary(e.data as ArrayBuffer)
    }

    const local = o.localStream
    if (local) for (const track of local.getTracks()) this.pc.addTrack(track, local)

    this.pc.ontrack = (e) => {
      const stream = e.streams[0]
      if (stream) o.onRemoteStream(stream)
    }
    this.pc.onicecandidate = (e) => {
      if (e.candidate && !this.closed) {
        const c = e.candidate
        o.signal({
          type: 'ice-candidate',
          candidate: { candidate: c.candidate, sdpMid: c.sdpMid, sdpMLineIndex: c.sdpMLineIndex, usernameFragment: c.usernameFragment },
        })
      }
    }
    this.pc.oniceconnectionstatechange = () => this.onIce()

    o.onState('connecting')
    this.armFail(o.connectTimeoutMs ?? 20_000)
  }

  async start(): Promise<void> {
    const offer = await this.pc.createOffer()
    await this.pc.setLocalDescription(offer)
    const d = this.pc.localDescription
    if (d?.sdp && !this.closed) this.o.signal({ type: 'offer', offer: { type: 'offer', sdp: d.sdp } })
  }

  async handleSignal(m: SignalIn): Promise<void> {
    if (this.closed) return
    switch (m.type) {
      case 'offer': {
        await this.pc.setRemoteDescription(m.offer)
        await this.flush()
        await this.pc.setLocalDescription()
        const d = this.pc.localDescription
        if (d?.sdp && !this.closed) this.o.signal({ type: 'answer', answer: { type: 'answer', sdp: d.sdp } })
        return
      }
      case 'answer':
        await this.pc.setRemoteDescription(m.answer)
        await this.flush()
        return
      case 'ice-candidate':
        if (this.pc.remoteDescription) await this.add(m.candidate)
        else this.pending.push(m.candidate)
        return
    }
  }

  async replaceVideoTrack(track: MediaStreamTrack): Promise<void> {
    const sender = this.pc.getSenders().find((s) => s.track === null || s.track.kind === 'video')
    await sender?.replaceTrack(track)
  }

  sendText(text: string): boolean {
    if (this.channel.readyState !== 'open') return false
    this.channel.send(text)
    return true
  }

  sendBinary(data: ArrayBuffer): boolean {
    if (this.channel.readyState !== 'open') return false
    this.channel.send(data)
    return true
  }

  /** Resolves immediately when the send buffer is below the low-water mark, else when it drains to it. */
  waitForDrain(): Promise<void> {
    if (this.channel.bufferedAmount <= this.channel.bufferedAmountLowThreshold) return Promise.resolve()
    return new Promise((resolve) => {
      this.channel.onbufferedamountlow = () => {
        this.channel.onbufferedamountlow = null
        resolve()
      }
    })
  }

  close(): void {
    if (this.closed) return
    this.closed = true
    this.clearFail()
    this.pc.onicecandidate = null
    this.pc.ontrack = null
    this.pc.oniceconnectionstatechange = null
    this.channel.onmessage = null
    try {
      this.channel.close()
    } catch {
      // already closed
    }
    this.pc.close()
  }

  private async flush(): Promise<void> {
    for (const candidate of this.pending.splice(0)) await this.add(candidate)
  }

  private async add(candidate: IceCandidateInit): Promise<void> {
    try {
      await this.pc.addIceCandidate(candidate)
    } catch {
      // a candidate from a connection we already replaced; harmless
    }
  }

  private onIce(): void {
    if (this.closed) return
    switch (this.pc.iceConnectionState) {
      case 'connected':
      case 'completed':
        this.clearFail()
        this.o.onState('connected')
        return
      case 'disconnected':
        this.o.onState('reconnecting')
        this.armFail(this.o.failAfterMs ?? 10_000)
        return
      case 'failed':
        this.clearFail()
        this.o.onState('failed')
        return
      default:
        return
    }
  }

  private armFail(ms: number): void {
    if (this.failTimer !== null) return
    this.failTimer = setTimeout(() => {
      this.failTimer = null
      if (!this.closed) this.o.onState('failed')
    }, ms)
  }

  private clearFail(): void {
    if (this.failTimer !== null) clearTimeout(this.failTimer)
    this.failTimer = null
  }
}
```

```ts
import { useCallback, useEffect, useRef, useState } from 'react'
import { DEFAULT_STUN } from '../../shared/constants'
import type { ClientMessage, IceServerConfig } from '../../shared/protocol'
import { Peer, type PeerState, type SignalIn } from '../lib/webrtc'
import { useLatest } from './useLatest'

interface Args {
  code: string
  send: (message: ClientMessage) => boolean
  localStream: MediaStream | null
  onChannelText: (text: string) => void
  onChannelBinary: (data: ArrayBuffer) => void
}

async function fetchIceServers(code: string): Promise<IceServerConfig[]> {
  try {
    const res = await fetch(`/api/ice?room=${code}`)
    if (res.ok) return ((await res.json()) as { iceServers: IceServerConfig[] }).iceServers
  } catch {
    // fall through to the public default
  }
  return [{ urls: DEFAULT_STUN }]
}

export function useWebRTC(args: Args) {
  const [remoteStream, setRemoteStream] = useState<MediaStream | null>(null)
  const [peerState, setPeerState] = useState<PeerState>('closed')
  const [channelOpen, setChannelOpen] = useState(false)
  const peerRef = useRef<Peer | null>(null)
  const generation = useRef(0)
  const chain = useRef<Promise<void>>(Promise.resolve())
  const latest = useLatest(args)

  const closePeer = useCallback(() => {
    generation.current += 1 // invalidates any build still waiting on the network
    peerRef.current?.close()
    peerRef.current = null
    setRemoteStream(null)
    setPeerState('closed')
    setChannelOpen(false)
  }, [])

  const build = useCallback(
    async (offer: boolean) => {
      closePeer()
      const gen = generation.current
      const a = latest.current
      if (!a.localStream) return
      const iceServers = await fetchIceServers(a.code)
      if (gen !== generation.current) return // superseded while fetching
      const peer = new Peer({
        iceServers,
        localStream: a.localStream,
        signal: (m) => void latest.current.send(m),
        onRemoteStream: (s) => {
          if (gen === generation.current) setRemoteStream(s)
        },
        onState: (s) => {
          if (gen === generation.current) setPeerState(s)
        },
        onChannelOpen: () => {
          if (gen === generation.current) setChannelOpen(true)
        },
        onChannelText: (t) => latest.current.onChannelText(t),
        onChannelBinary: (b) => latest.current.onChannelBinary(b),
      })
      peerRef.current = peer
      if (offer) await peer.start()
    },
    [closePeer, latest],
  )

  /** Signals are handled strictly in order: an offer's build must finish before its candidates arrive. */
  const enqueue = useCallback((job: () => Promise<void>) => {
    chain.current = chain.current.then(job).catch(() => setPeerState('failed'))
  }, [])

  const buildAndOffer = useCallback(() => enqueue(() => build(true)), [build, enqueue])

  const onSignal = useCallback(
    (m: SignalIn) =>
      enqueue(async () => {
        if (m.type === 'offer') await build(false)
        await peerRef.current?.handleSignal(m)
      }),
    [build, enqueue],
  )

  // Swap the outgoing video track in place when the camera is flipped (no renegotiation).
  const localStream = args.localStream
  useEffect(() => {
    const track = localStream?.getVideoTracks()[0]
    if (track) void peerRef.current?.replaceVideoTrack(track)
  }, [localStream])

  useEffect(() => closePeer, [closePeer])

  return {
    remoteStream,
    peerState,
    channelOpen,
    buildAndOffer,
    onSignal,
    closePeer,
    sendText: useCallback((s: string) => peerRef.current?.sendText(s) ?? false, []),
    sendBinary: useCallback((b: ArrayBuffer) => peerRef.current?.sendBinary(b) ?? false, []),
    waitForDrain: useCallback(() => peerRef.current?.waitForDrain() ?? Promise.resolve(), []),
  }
}
```

```tsx
import { useEffect, useRef, type RefObject } from 'react'

interface Props {
  stream: MediaStream | null
  label: string
  mirrored: boolean
  testId: string
  videoRef?: RefObject<HTMLVideoElement | null>
}

export function CameraFrame({ stream, label, mirrored, testId, videoRef }: Props) {
  const own = useRef<HTMLVideoElement>(null)
  const ref = videoRef ?? own
  useEffect(() => {
    const video = ref.current
    if (!video) return
    video.srcObject = stream
    if (stream) void video.play().catch(() => {}) // autoplay can be refused until a gesture; the element is muted so it normally isn't
  }, [stream, ref])
  return (
    <figure className="relative aspect-[4/3] w-full overflow-hidden border-4 border-black bg-black">
      <video
        ref={ref}
        data-testid={testId}
        aria-label={`${label} camera`}
        autoPlay
        muted
        playsInline
        className="h-full w-full object-cover"
        style={{ transform: mirrored ? 'scaleX(-1)' : undefined }}
      />
      <figcaption className="absolute left-0 top-0 border-b-4 border-r-4 border-black bg-yellow-300 px-2 py-1 text-sm font-black uppercase">
        {label}
      </figcaption>
    </figure>
  )
}
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react'
import { useNavigate } from 'react-router'

export function CreateBooth() {
  const navigate = useNavigate()
  const [failed, setFailed] = useState(false)
  const started = useRef(false)

  const create = useCallback(async () => {
    setFailed(false)
    try {
      const res = await fetch('/api/rooms', { method: 'POST' })
      if (!res.ok) throw new Error(String(res.status))
      const { code } = (await res.json()) as { code: string }
      void navigate(`/b/${code}`, { replace: true })
    } catch {
      setFailed(true)
    }
  }, [navigate])

  useEffect(() => {
    if (started.current) return // StrictMode runs effects twice in dev; make one room, not two
    started.current = true
    void create()
  }, [create])

  return (
    <main className="mx-auto flex min-h-dvh max-w-xl flex-col justify-center gap-4 p-6">
      {failed ? (
        <>
          <h1 className="text-3xl font-black">COULDN'T BUILD YOUR BOOTH.</h1>
          <button type="button" onClick={() => void create()} className="border-4 border-black px-4 py-3 font-black">
            TRY AGAIN
          </button>
        </>
      ) : (
        <h1 className="text-3xl font-black">BUILDING YOUR BOOTH…</h1>
      )}
    </main>
  )
}
```

```tsx
import { useState, type FormEvent } from 'react'
import { useNavigate } from 'react-router'
import { normalizeRoomInput } from '../../shared/room-code'
import { getParticipantId } from '../lib/session'

export function JoinForm() {
  const navigate = useNavigate()
  const [value, setValue] = useState('')
  const [error, setError] = useState<string | null>(null)
  const [busy, setBusy] = useState(false)

  async function submit(e: FormEvent) {
    e.preventDefault()
    const code = normalizeRoomInput(value)
    if (!code) {
      setError('THAT CODE LOOKS WRONG.')
      return
    }
    setBusy(true)
    setError(null)
    try {
      const res = await fetch(`/api/rooms/${code}`, { headers: { 'x-participant-id': getParticipantId(code) } })
      if (res.status === 404) setError('NO BOOTH WITH THAT CODE.')
      else if (!res.ok) setError('COULD NOT CHECK THAT BOOTH. TRY AGAIN.')
      else if (((await res.json()) as { full: boolean }).full) setError('BOOTH FULL.')
      else void navigate(`/b/${code}`)
    } catch {
      setError('COULD NOT REACH THE SERVER.')
    } finally {
      setBusy(false)
    }
  }

  return (
    <form onSubmit={(e) => void submit(e)} className="flex flex-col gap-3">
      <label htmlFor="join-code" className="font-black">
        ENTER CODE
      </label>
      <input
        id="join-code"
        value={value}
        onChange={(e) => setValue(e.target.value)}
        autoComplete="off"
        autoCapitalize="characters"
        spellCheck={false}
        className="border-4 border-black px-3 py-3 text-2xl font-black uppercase tracking-widest"
      />
      {error && (
        <p role="alert" className="font-black">
          {error}
        </p>
      )}
      <button type="submit" disabled={busy} className="border-4 border-black px-4 py-3 font-black">
```

```tsx
import { useMemo, useRef } from 'react'
import { useParams } from 'react-router'
import { otherSlot, type ServerMessage, type SlotName } from '../../shared/protocol'
import { normalizeRoomInput } from '../../shared/room-code'
import { useCamera } from '../hooks/useCamera'
import { useLatest } from '../hooks/useLatest'
import { useRoom } from '../hooks/useRoom'
import { useRoomGate } from '../hooks/useRoomGate'
import { useWebRTC } from '../hooks/useWebRTC'
import type { CameraError } from '../lib/camera'
import { getParticipantId } from '../lib/session'
import { CameraGate } from '../screens/CameraGate'
import { Lobby } from '../screens/Lobby'
import { Problem, type ProblemKind } from '../screens/Problem'
import { Stage } from '../screens/Stage'

const cameraProblem: Record<CameraError, ProblemKind> = {
  blocked: 'camera-blocked',
  unavailable: 'camera-unavailable',
  unsupported: 'camera-unsupported',
}

function closedProblem(code: number | null): ProblemKind {
  switch (code) {
    case 4403:
      return 'room-full'
    case 4404:
      return 'not-found'
    case 4000:
      return 'expired'
    case 4001:
      return 'replaced'
    default:
      return 'connection-lost'
  }
}

export function Booth() {
  const { code: raw } = useParams()
  const code = normalizeRoomInput(raw ?? '')
  return code ? <BoothInner code={code} /> : <Problem kind="not-found" />
}

function BoothInner({ code }: { code: string }) {
  const participantId = useMemo(() => getParticipantId(code), [code])
  const gate = useRoomGate(code, participantId)
  const camera = useCamera()
  const localVideo = useRef<HTMLVideoElement | null>(null)
  const slot = useRef<SlotName | null>(null)

  // Server events → WebRTC. `latest` keeps this closure fresh without reconnecting the socket.
  const onMessage = useLatest((m: ServerMessage, isSnapshot: boolean) => {
    switch (m.type) {
      case 'room-state':
        slot.current = m.state.you
        // Slot a re-offers when it joins a booth where b is already present (b never offers).
        if (isSnapshot && m.state.you === 'a' && m.state.participants.b?.connected) rtc.buildAndOffer()
        return
      case 'partner-joined':
        if (slot.current === 'a') rtc.buildAndOffer()
        else rtc.closePeer() // b waits for a's offer and rebuilds on it
        return
      case 'partner-left':
        rtc.closePeer()
        return
      case 'offer':
      case 'answer':
      case 'ice-candidate':
        rtc.onSignal(m)
        return
      default:
        return
    }
  })

  const room = useRoom({
    code,
    participantId,
    enabled: gate === 'ok' && camera.stream !== null,
    onMessage: (m, isSnapshot) => onMessage.current(m, isSnapshot),
  })
  const rtc = useWebRTC({
    code,
    send: room.send,
    localStream: camera.stream,
    onChannelText: () => {},
    onChannelBinary: () => {},
  })

  if (gate === 'checking') return <main className="p-6 font-black">CHECKING BOOTH…</main>
  if (gate === 'not-found') return <Problem kind="not-found" />
  if (gate === 'full') return <Problem kind="room-full" />
  if (gate === 'error') return <Problem kind="gate-error" onRetry={() => location.reload()} />

  if (camera.error) return <Problem kind={cameraProblem[camera.error]} onRetry={() => void camera.start()} />
  if (!camera.stream) return <CameraGate onStart={() => void camera.start()} />

  if (room.status === 'closed') return <Problem kind={closedProblem(room.closeCode)} onRetry={room.reconnect} />
  if (!room.state) return <main className="p-6 font-black">CONNECTING…</main>

  const partner = room.state.participants[otherSlot(room.state.you)]
  if (!partner?.connected) {
    return <Lobby code={code} shareUrl={`${location.origin}/b/${code}`} stream={camera.stream} mirrored={camera.mirrored} />
  }
  if (rtc.peerState === 'failed') return <Problem kind="webrtc-failed" onRetry={room.reconnect} />

  return (
    <Stage
      stream={camera.stream}
      mirrored={camera.mirrored}
      remoteStream={rtc.remoteStream}
      remoteMirrored
      peerState={rtc.peerState}
      localVideoRef={localVideo}
    />
  )
}
```

```ts
import { expect, type Browser, type Page } from '@playwright/test'

export const BASE = 'http://localhost:5199'

/** One isolated browser context = one person (own storage, own participant id). */
export async function newPlayer(browser: Browser, opts: { viewport?: { width: number; height: number } } = {}): Promise<Page> {
  const context = await browser.newContext({ baseURL: BASE, permissions: ['camera'], ...opts })
  return context.newPage()
}

export async function createBooth(page: Page): Promise<string> {
  await page.goto('/create')
  await page.waitForURL(/\/b\/[A-HJ-KM-NP-Z2-9]{6}$/)
  return page.url()
}

export async function enterBooth(page: Page): Promise<void> {
  await page.getByRole('button', { name: /turn on camera/i }).click()
}

export async function expectRemoteVideo(page: Page): Promise<void> {
  await expect
    .poll(
      () =>
        page.evaluate(() => {
          const v = document.querySelector<HTMLVideoElement>('[data-testid="remote-video"]')
          return !!v && v.videoWidth > 0 && v.currentTime > 0
        }),
      { message: 'remote video should be decoding frames' },
    )
    .toBe(true)
}
```
